1.1 The Golden Rule

Build the simplest possible thing that proves the control plane works.

No AI/ML in Pilot A. No Turso. No WebSockets. No microservices. No Kubernetes. One database. One runtime. One orchestration layer. One source of truth.

Everything else is premature complexity.

4.1 Database Schema (Drizzle ORM)

typescript
// src/db/schema.ts
import { 
  pgTable, 
  varchar, 
  decimal, 
  timestamp, 
  integer, 
  boolean, 
  jsonb,
  pgEnum,
  index,
  uniqueIndex
} from "drizzle-orm/pg-core";

// Enums
export const statusEnum = pgEnum("status", [
  "RECEIVED", "VALIDATED", "POLICY_APPROVED", 
  "POLICY_BLOCKED", "ROUTE_SELECTED", "FINALITY_F0",
  "FINALITY_F1", "FINALITY_F2", "FINALITY_F3",
  "FINALITY_F4", "FINALITY_F5", "FINALITY_F6",
  "FINALITY_F7", "SETTLED", "EXCEPTION", "SAFE_HALTED"
]);

export const severityEnum = pgEnum("severity", [
  "LOW", "MEDIUM", "HIGH", "CRITICAL"
]);

// Institutions
export const institutions = pgTable("institutions", {
  id: varchar("id", { length: 50 }).primaryKey(),
  legalName: varchar("legal_name", { length: 255 }).notNull(),
  jurisdiction: varchar("jurisdiction", { length: 10 }).notNull(),
  status: varchar("status", { length: 20 }).notNull().default("ACTIVE"),
  participantType: varchar("participant_type", { length: 50 }).notNull(),
  createdAt: timestamp("created_at").notNull().defaultNow(),
  updatedAt: timestamp("updated_at").notNull().defaultNow(),
}, (table) => ({
  jurisdictionIdx: index("jurisdiction_idx").on(table.jurisdiction),
}));

// Instructions
export const instructions = pgTable("instructions", {
  id: varchar("id", { length: 50 }).primaryKey(),
  institutionId: varchar("institution_id", { length: 50 })
    .notNull()
    .references(() => institutions.id),
  corridor: varchar("corridor", { length: 50 }).notNull(),
  sourceCurrency: varchar("source_currency", { length: 3 }).notNull(),
  targetCurrency: varchar("target_currency", { length: 3 }).notNull(),
  amount: decimal("amount", { precision: 20, scale: 2 }).notNull(),
  debtorName: varchar("debtor_name", { length: 255 }).notNull(),

// src/app/api/v1/instructions/route.ts
import { NextRequest, NextResponse } from "next/server";
import { z } from "zod";
import { db } from "@/db";
import { instructions, outbox } from "@/db/schema";
import { inngest } from "@/inngest/client";

const submitInstructionSchema = z.object({
  instructionId: z.string().min(1).max(50),
  institutionId: z.string().min(1).max(50),
  corridor: z.string().min(1).max(50),
  sourceCurrency: z.string().length(3),
  targetCurrency: z.string().length(3),
  amount: z.number().positive(),
  debtor: z.object({
    name: z.string().min(1).max(255),
    account: z.string().min(1).max(50),
  }),
  creditor: z.object({
    name: z.string().min(1).max(255),
    account: z.string().min(1).max(50),
  }),
  purpose: z.string().min(1).max(50),
  requestedExecutionDate: z.string(),
});

export async function POST(request: NextRequest) {
  // 1. Authenticate (mTLS + API key)
  const auth = await authenticateRequest(request);
  if (!auth.valid) {
    return NextResponse.json(
      { error: { code: "UNAUTHORIZED", message: auth.message } },
      { status: 401 }
    );
  }

  // 2. Validate request
  const body = await request.json();
  const parsed = submitInstructionSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json(
      { error: { code: "INVALID_SCHEMA", message: parsed.error.message } },
      { status: 400 }
    );
  }

  const data = parsed.data;

  // 3. Check idempotency
  const idempotencyKey = request.headers.get("X-Idempotency-Key");
  if (!idempotencyKey) {
    return NextResponse.json(
      { error: { code: "MISSING_IDEMPOTENCY_KEY" } },
      { status: 400 }
    );
  }

  // 4. Write to database in a single transaction
  const result = await db.transaction(async (tx) => {
    // Insert instruction
    const [instruction] = await tx
      .insert(instructions)
      .values({
        id: data.instructionId,
        institutionId: data.institutionId,
        corridor: data.corridor,
        sourceCurrency: data.sourceCurrency,
        targetCurrency: data.targetCurrency,
        amount: data.amount.toString(),
        debtorName: data.debtor.name,
        creditorName: data.creditor.name,
        purpose: data.purpose,
        status: "RECEIVED",
        policyVersion: "POLICY-v1.2.3",
      })
      .returning();

    // Insert outbox event
    await tx.insert(outbox).values({
      id: crypto.randomUUID(),
      aggregateType: "INSTRUCTION",
      aggregateId: data.instructionId,
      eventType: "INSTRUCTION_RECEIVED",
      payload: { instructionId: data.instructionId },
      status: "PENDING",
    });

    return instruction;
  });

  // 5. Send event to Inngest (after transaction commits)
  await inngest.send({
    name: "instruction/received",
    data: { instructionId: data.instructionId },
  });

  // 6. Return response
  return NextResponse.json({
    instructionId: result.id,
    status: result.status,
    policyVersion: result.policyVersion,
    finalityState: "F0",
    createdAt: result.createdAt.toISOString(),
  });
}

4.3 Inngest Workflows

typescript
// src/inngest/functions/process-instruction.ts
import { inngest } from "../client";
import { db } from "@/db";
import { instructions, outbox, policyDecisions } from "@/db/schema";
import { eq } from "drizzle-orm";

export const processInstruction = inngest.createFunction(
  {
    id: "process-instruction",
    retries: 3,
    triggers: { event: "instruction/received" },
  },
  async ({ event, step }) => {
    const { instructionId } = event.data;

    // Step 1: Validate
    const instruction = await step.run("validate", async () => {
      const result = await db.query.instructions.findFirst({
        where: eq(instructions.id, instructionId),
      });
      if (!result) throw new Error("Instruction not found");
      return result;
    });

    // Step 2: Check policy
    const policyResult = await step.run("check-policy", async () => {
      const decision = await evaluatePolicy(instruction);
      await db.insert(policyDecisions).values({
        id: crypto.randomUUID(),
        instructionId,
        policyVersion: "POLICY-v1.2.3",
        decision: decision.approved ? "APPROVED" : "BLOCKED",
        reasonCodes: decision.reasonCodes,
        actor: "MITHQAL-CORE",
      });
      return decision;
    });

    if (!policyResult.approved) {
      await step.run("block-instruction", async () => {
        await db
          .update(instructions)
          .set({ status: "POLICY_BLOCKED" })
          .where(eq(instructions.id, instructionId));
      });
      return { status: "BLOCKED" };
    }

    // Step 3: Select route
    const route = await step.run("select-route", async () => {
      return selectBestRoute(instruction);
    });

    // Step 4: Update instruction
    await step.run("update-instruction", async () => {
      await db
        .update(instructions)
        .set({ status: "ROUTE_SELECTED" })
        .where(eq(instructions.id, instructionId));
    });

    return { status: "ROUTE_SELECTED", route };
  }
);

import { NextResponse } from "next/server";

export default auth((req) => {
  const { pathname } = req.nextUrl;
  const isPublic = pathname.startsWith("/api/auth") || pathname === "/";

  if (isPublic) return NextResponse.next();

  if (!req.auth) {
    return NextResponse.redirect(new URL("/login", req.url));

  // RBAC check
  const requiredRole = getRequiredRole(pathname);
  if (requiredRole && req.auth.user.role !== requiredRole) {
    return NextResponse.json(
      { error: { code: "FORBIDDEN" } },
      { status: 403 }
    );
  }

  return NextResponse.next();
});
